   415→/** Metadata about the file type used for downloads and exoplayer hint,
   416→ * if you respond with the wrong one the file will fail to download or be played */

   418→    /** Single stream of bytes no matter the actual file type */

   420→
   421→    /** Split into several .ts files, has support for encrypted m3u8s */

   423→
   424→    /** Like m3u8 but uses xml, currently no download support */
   425→    DASH,
   426→
   427→    /** No support at the moment */

   446→private fun inferTypeFromUrl(url: String): ExtractorLinkType {
   447→    val path = try {
   448→        Url(url).encodedPath.decodeURLPart()
   449→    } catch (_: Throwable) {
   450→        // don't log magnet links as errors
   451→        null
   452→    }
   453→    return when {
   454→        path?.endsWith(".m3u8") == true -> ExtractorLinkType.M3U8
   455→        path?.endsWith(".mpd") == true -> ExtractorLinkType.DASH
   456→        path?.endsWith(".torrent") == true -> ExtractorLinkType.TORRENT
   457→        url.startsWith("magnet:") -> ExtractorLinkType.MAGNET
   458→        else -> ExtractorLinkType.VIDEO
   459→    }
   460→}

   931→/**
   932→ * Tries to load the appropriate extractor based on link, returns true if any extractor is loaded.
   933→ * */
   934→@Throws(CancellationException::class)
   935→suspend fun loadExtractor(
   936→    url: String,
   937→    referer: String? = null,
   938→    subtitleCallback: (SubtitleFile) -> Unit,
   939→    callback: (ExtractorLink) -> Unit
   940→): Boolean {
   941→    // Ensure this coroutine has not timed out
   942→    coroutineScope { ensureActive() }
   943→
   944→    val currentUrl = unshortenLinkSafe(url)
   945→    val compareUrl = currentUrl.lowercase().replace(schemaStripRegex, "")
   946→
   947→    // Iterate in reverse order so the new registered ExtractorApi takes priority
   948→    for (index in extractorApis.lastIndex downTo 0) {
   949→        val extractor = extractorApis[index]
   950→        if (compareUrl.startsWith(extractor.mainUrl.replace(schemaStripRegex, ""))) {
   951→            try {
   952→                extractor.getUrl(currentUrl, referer, subtitleCallback, callback)
   953→            } catch (e: Exception) {
   954→                logError(e)
   955→                // Rethrow if we have timed out
   956→                if (e is CancellationException) {
   957→                    throw e
   958→                }
   959→            }
   960→            return true
   961→        }
   962→    }
   963→
   964→    // this is to match mirror domains - like example.com, example.net
   965→    for (index in extractorApis.lastIndex downTo 0) {
   966→        val extractor = extractorApis[index]
   967→        if (Levenshtein.partialRatio(
   968→                extractor.mainUrl,
   969→                currentUrl
   970→            ) > 80
   971→        ) {
   972→            try {
   973→                extractor.getUrl(currentUrl, referer, subtitleCallback, callback)
   974→            } catch (e: Exception) {
   975→                logError(e)
   976→                // Rethrow if we have timed out
   977→                if (e is CancellationException) {
   978→                    throw e
   979→                }
   980→            }
   981→            return true
   982→        }
   983→    }
   984→
   985→    return false
   986→}

  1367→suspend fun getPostForm(requestUrl: String, html: String): String? {
  1368→    val document = Ksoup.parse(html)
  1369→    val inputs = document.select("Form > input")
  1370→    if (inputs.size < 4) return null
  1371→    var op: String? = null
  1372→    var id: String? = null
  1373→    var mode: String? = null
  1374→    var hash: String? = null
  1375→
  1376→    for (input in inputs) {
  1377→        val value = input.attr("value")
  1378→        when (input.attr("name")) {
  1379→            "op" -> op = value
  1380→            "id" -> id = value
  1381→            "mode" -> mode = value
  1382→            "hash" -> hash = value
  1383→            else -> Unit
  1384→        }
  1385→    }
  1386→    if (op == null || id == null || mode == null || hash == null) {
  1387→        return null
  1388→    }
  1389→    delay(5000) // ye this is needed, wont work with 0 delay
  1390→
  1391→    return app.post(
  1392→        requestUrl,
  1393→        headers = mapOf(
  1394→            "content-type" to "application/x-www-form-urlencoded",
  1395→            "referer" to requestUrl,
  1396→            "user-agent" to USER_AGENT,
  1397→            "accept" to "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.9"
  1398→        ),
  1399→        data = mapOf("op" to op, "id" to id, "mode" to mode, "hash" to hash)
  1400→    ).text
  1401→}

  1403→fun ExtractorApi.fixUrl(url: String): String {
  1404→    if (url.startsWith("http") ||
  1405→        // Do not fix JSON objects when passed as urls.
  1406→        url.startsWith("{\"")
  1407→    ) {
  1408→        return url
  1409→    }
  1410→    if (url.isEmpty()) {
  1411→        return ""
  1412→    }
  1413→
  1414→    val startsWithNoHttp = url.startsWith("//")
  1415→    if (startsWithNoHttp) {
  1416→        return "https:$url"
  1417→    } else {
  1418→        if (url.startsWith('/')) {
  1419→            return mainUrl + url
  1420→        }
  1421→        return "$mainUrl/$url"
  1422→    }
  1423→}
